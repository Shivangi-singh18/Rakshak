# requirements.txt — Rakshak API backend (FastAPI + SQLAlchemy + SQLite)
#
# Python: 3.11 – 3.14  (verified on Python 3.14.7, Windows venv)
# Install:  python -m pip install -r requirements.txt
# Run:      uvicorn main:app --reload
#
# All versions below are exact pins (==) for reproducible, production-ready
# installs. The direct dependencies are the current stable releases as of
# 2026-10-04; transitive runtime dependencies are pinned too, so a fresh
# install resolves identically on every machine.
#
# ---------------------------------------------------------------------------
# SQLite compatibility ("sqlite3")
# ---------------------------------------------------------------------------
# sqlite3 is part of the Python *standard library* — CPython ships it by
# default and SQLAlchemy's SQLite dialect (sqlite:///./rakshak.db in
# database.py) uses it directly. It must NOT appear as a pip requirement
# here: the "sqlite3" name on PyPI is an unrelated placeholder project.
# To confirm your interpreter has it (and the underlying library):
#     python -c "import sqlite3; print(sqlite3.sqlite_version)"
# No third-party driver (no pysqlite, no aiosqlite) is needed for this app.

# ---------------------------------------------------------------------------
# Direct dependencies
# ---------------------------------------------------------------------------
fastapi==0.142.2             # Web framework (requires Python >=3.10)
uvicorn==0.54.0               # ASGI server; add [standard] extras for prod: uvicorn[standard]==0.54.0
SQLAlchemy==2.1.3             # ORM + engine; SQLite via stdlib sqlite3 (requires Python >=3.11)
pydantic==2.13.5              # Request/response schemas (schemas.py)
reportlab==5.0.1              # PDF generation (pure-Python wheel, no compiler needed)
python-multipart==0.0.32      # Form/file upload parsing for FastAPI (FastAPI >=0.119 requires it)
